type journal struct {
	f          *os.File
	calls      int
	bodyHashes []string
}

func (j *journal) add(v map[string]any) error {
	v["at"] = time.Now().UTC()
	if json.NewEncoder(j.f).Encode(v) != nil || j.f.Sync() != nil {
		return errors.New("journal_write_failed")
	}
	return nil
}

type observer struct {
	next  http.RoundTripper
	j     *journal
	base  string
	room  string
	actor string
}

func (o *observer) RoundTrip(req *http.Request) (*http.Response, error) {
	if req.URL.Scheme+"://"+req.URL.Host != o.base || req.Method != "POST" || req.URL.RawQuery != "" {
		return nil, errors.New("probe_network_scope_rejected")
	}
	raw, err := io.ReadAll(io.LimitReader(req.Body, 65537))
	if err != nil || len(raw) > 65536 {
		return nil, errors.New("probe_request_invalid")
	}
	req.Body = io.NopCloser(bytes.NewReader(raw))
	form, err := url.ParseQuery(string(raw))
	if err != nil {
		return nil, errors.New("probe_form_invalid")
	}
	operation := ""
	switch req.URL.Path {
	case "/group/user/query.json":
		if len(form) != 1 || form.Get("groupId") != o.room {
			return nil, errors.New("probe_group_read_scope_rejected")
		}
		operation = "group_members_read"
	case "/message/group/publish.json":
		if form.Get("toGroupId") != o.room || len(form["toGroupId"]) != 1 || form.Get("fromUserId") != o.actor || form.Get("objectName") != "RC:CmdMsg" || form.Get("isPersisted") != "0" {
			return nil, errors.New("probe_command_scope_rejected")
		}
		operation = "reaction_command"
	default:
		return nil, errors.New("probe_route_rejected")
	}
	o.j.calls++
	if err = o.j.add(map[string]any{"operation": operation, "state": "network_started", "request_path": req.URL.Path, "form_sha256": hash(raw)}); err != nil {
		return nil, err
	}
	resp, err := o.next.RoundTrip(req)
	if err != nil {
		_ = o.j.add(map[string]any{"operation": operation, "state": "network_unknown"})
		return nil, err
	}
	body, err := io.ReadAll(io.LimitReader(resp.Body, 262145))
	resp.Body.Close()
	if err != nil || len(body) > 262144 {
		_ = o.j.add(map[string]any{"operation": operation, "state": "response_unknown", "http_status": resp.StatusCode})
		return nil, errors.New("probe_response_unavailable")
	}
	resp.Body = io.NopCloser(bytes.NewReader(body))
	o.j.bodyHashes = append(o.j.bodyHashes, hash(body))
	if err = o.j.add(map[string]any{"operation": operation, "state": "response_observed", "http_status": resp.StatusCode, "raw_body_bytes": len(body), "raw_body_sha256": hash(body)}); err != nil {
		return nil, err
	}
	return resp, nil
}

type scoped struct {
	p           *transport.RongCloud
	j           *journal
	expected    domain.ReactionReceipt
	text        domain.Message
	sent        bool
	admin       *pgx.Conn
	currentRoom domain.Room
}

func (s *scoped) Session(context.Context, domain.Principal) (transport.Session, error) {
	return transport.Session{}, domain.ErrForbidden
}
func (s *scoped) CreateGroup(context.Context, domain.Room, []string) error {
	return domain.ErrForbidden
}
func (s *scoped) Publish(_ context.Context, m domain.Message) (transport.Delivery, error) {
	if m.ID != s.text.ID || m.RoomID != s.text.RoomID || m.AuthorID != s.text.AuthorID || m.Content != s.text.Content || m.Seq != s.text.Seq {
		return transport.Delivery{}, domain.ErrInvalid
	}
	if err := s.j.add(map[string]any{"operation": "local_text", "state": "deliberately_blocked_by_fixture_scope", "message_id": m.ID, "external_calls": 0, "note": "The scoped probe rejects all ordinary text publication using ErrForbidden; outbox membership_revoked is this test suppression code, not an actual membership removal."}); err != nil {
		return transport.Delivery{}, err
	}
	return transport.Delivery{}, domain.ErrForbidden
}
func (s *scoped) NotifyReaction(ctx context.Context, r domain.ReactionReceipt) (transport.Delivery, error) {
	if r != s.expected || s.sent {
		return transport.Delivery{}, domain.ErrInvalid
	}
	s.sent = true
	var live domain.Room
	e := s.admin.QueryRow(ctx, "SELECT id::text,workspace_id::text,title,kind,version,scope_epoch,stopped FROM public.rooms WHERE id=$1", r.RoomID).Scan(&live.ID, &live.WorkspaceID, &live.Title, &live.Kind, &live.Version, &live.ScopeEpoch, &live.Stopped)
	if e != nil || live != s.currentRoom {
		return transport.Delivery{}, domain.ErrStopped
	}
	if e = s.j.add(map[string]any{"operation": "pre_command_policy_read", "state": "current_policy_matches_frozen_plan", "version": live.Version, "scope_epoch": live.ScopeEpoch, "stopped": live.Stopped}); e != nil {
		return transport.Delivery{}, e
	}
	if err := s.j.add(map[string]any{"operation": "reaction_command", "state": "intent_fsynced", "room_id": r.RoomID, "message_id": r.MessageID, "principal_id": r.PrincipalID, "version": r.Version, "attempt_budget": 1}); err != nil {
		return transport.Delivery{}, err
	}
	d, err := s.p.NotifyReaction(ctx, r)
	state := "provider_accepted"
	unknown := false
	code := d.Code
	if err != nil {
		state = "stopped_no_retry"
		unknown = true
		var pe *transport.ProviderError
		if errors.As(err, &pe) {
			unknown = pe.Unknown
			code = pe.Code
		}
	}
	v := map[string]any{"operation": "reaction_command", "state": state, "unknown": unknown, "code": code}
	if err == nil {
		v["provider_receipt"] = d
	}
	if e := s.j.add(v); e != nil {
		return d, e
	}
	return d, err
}

func publicSnapshot(ctx context.Context, c *pgx.Conn, room string) ([]byte, error) {
	var out []byte
	err := c.QueryRow(ctx, `SELECT jsonb_build_object('room',(SELECT to_jsonb(r) FROM public.rooms r WHERE id=$1),'members',(SELECT jsonb_agg(to_jsonb(m) ORDER BY principal_id) FROM public.room_members m WHERE room_id=$1),'messages',(SELECT jsonb_agg(to_jsonb(m) ORDER BY seq) FROM public.messages m WHERE room_id=$1),'events',(SELECT jsonb_agg(to_jsonb(e) ORDER BY id) FROM public.events e WHERE room_id=$1),'outbox',(SELECT jsonb_agg(to_jsonb(o) ORDER BY o.id) FROM public.transport_outbox o JOIN public.events e ON e.id=o.event_id WHERE e.room_id=$1))`, room).Scan(&out)
	return out, err
}
func isolatedSnapshot(ctx context.Context, s *store.Store) ([]byte, error) {
	var out []byte
	err := s.Pool.QueryRow(ctx, `SELECT jsonb_build_object('messages',(SELECT jsonb_agg(to_jsonb(m) ORDER BY seq) FROM messages m),'actions',(SELECT jsonb_agg(to_jsonb(a) ORDER BY action_id) FROM actions a),'events',(SELECT jsonb_agg(to_jsonb(e) ORDER BY id) FROM events e),'outbox',(SELECT jsonb_agg(to_jsonb(o) ORDER BY id) FROM transport_outbox o),'reactions',(SELECT jsonb_agg(to_jsonb(r)) FROM message_reactions r))`).Scan(&out)
	return out, err
}
func readMembers(ctx context.Context, cfg map[string]string, group string) ([]string, error) {
	nonceBytes := make([]byte, 16)
	if _, err := rand.Read(nonceBytes); err != nil {
		return nil, err
	}
	nonce := hex.EncodeToString(nonceBytes)
	ts := strconv.FormatInt(time.Now().Unix(), 10)
	sig := sha1.Sum([]byte(cfg["RONGCLOUD_APP_SECRET"] + nonce + ts))
	req, err := http.NewRequestWithContext(ctx, "POST", cfg["RONGCLOUD_API_URL"]+"/group/user/query.json", strings.NewReader(url.Values{"groupId": {group}}.Encode()))
	if err != nil {
		return nil, err
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	req.Header.Set("App-Key", cfg["RONGCLOUD_APP_KEY"])
	req.Header.Set("Nonce", nonce)
	req.Header.Set("Timestamp", ts)
	req.Header.Set("Signature", hex.EncodeToString(sig[:]))
	client := &http.Client{Timeout: 8 * time.Second, CheckRedirect: func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }}
	resp, err := client.Do(req)
	if err != nil {
		return nil, errors.New("group_read_failed")
	}
	defer resp.Body.Close()
	var result struct {
		Code  int `json:"code"`
		Users []struct {
			ID string `json:"id"`
		} `json:"users"`
	}
	if resp.StatusCode != 200 || json.NewDecoder(resp.Body).Decode(&result) != nil || result.Code != 200 {
		return nil, errors.New("group_read_rejected")
	}
	ids := []string{}
	for _, u := range result.Users {
		ids = append(ids, u.ID)
	}
	return ids, nil
}
func hash(b []byte) string { v := sha256.Sum256(b); return hex.EncodeToString(v[:]) }
func save(path string, v any) error {
	b, e := json.MarshalIndent(v, "", "  ")
	if e != nil {
		return e
	}
	return saveRaw(path, append(b, '\n'))
}
func saveRaw(path string, b []byte) error {
	f, e := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if e != nil {
		return e
	}
	defer f.Close()
	if _, e = f.Write(b); e != nil {
		return e
	}
	if err := f.Sync(); err != nil {
		return err
	}
	return syncDir(filepath.Dir(path))
}

func syncDir(path string) error {
	f, err := os.Open(path)
	if err != nil {
		return err
	}
	defer f.Close()
	return f.Sync()
}
